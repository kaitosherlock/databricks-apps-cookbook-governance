# Modified 2026-09-21: make the build work on the Databricks Apps runtime.
#
# databricks-connect 16.2+ ships wheels marked `Requires-Python ==3.12.*`, and the
# Apps build image is not 3.12, so `databricks-connect~=18.1` has no installable
# candidate there and pip fails the whole build. The marker below installs it only
# on 3.12, so the app builds everywhere.
#
# Only views/compute_connect.py and views/compute_connect_serverless.py import it.
# Those two pages raise ImportError when opened on a runtime without it; every
# other page, including Governance, is unaffected because Streamlit's
# st.navigation imports a page's module only when you navigate to it.
databricks-connect~=18.1; python_version == "3.12"

databricks-sdk[openai]~=0.105
databricks-sql-connector~=4.2
pandas~=3.0
psycopg[binary]~=3.3
streamlit~=1.56
streamlit-folium~=0.27
